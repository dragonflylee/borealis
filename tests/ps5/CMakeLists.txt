cmake_minimum_required(VERSION 3.18)
project(borealis_ps5_tests LANGUAGES CXX)
enable_testing()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
file(GLOB contract_tests CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/test_*.py")
foreach(test IN LISTS contract_tests)
    get_filename_component(name "${test}" NAME_WE)
    add_test(NAME ${name} COMMAND ${Python3_EXECUTABLE} "${test}")
endforeach()
if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/hdr_modes.cpp")
    add_executable(hdr_modes hdr_modes.cpp)
    target_compile_features(hdr_modes PRIVATE cxx_std_17)
    target_compile_options(hdr_modes PRIVATE -Wall -Wextra -Werror -UNDEBUG -fsanitize=address,undefined)
    target_link_options(hdr_modes PRIVATE -fsanitize=address,undefined)
    target_include_directories(hdr_modes PRIVATE ../../library/include)
    add_test(NAME hdr_modes COMMAND hdr_modes)
endif()
